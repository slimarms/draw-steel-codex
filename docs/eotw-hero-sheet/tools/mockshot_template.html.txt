        '<div class="acts"><button class="btn" data-act="dismiss">Free strike</button><button class="btn" data-act="dismiss">Let it go</button></div></div>';
    }
    if (state.overlay === "turn") {
      return '<div class="ov-turn" role="status"><div><div class="t">Your turn</div><div class="d">' + esc(h.name) + " acts now. The sheet stays open until you close it.</div></div>" +
        '<button class="btn" data-act="close">Close sheet <kbd>Esc</kbd></button><button class="btn" data-act="dismiss">Keep reading</button></div>';
    }
    if (state.overlay === "closed") {
      return '<div class="ov-closed"><div>The sheet closed (Escape). In the app you would be back in ' + (state.ctx === "town" ? "the Hero's Guild" : "the encounter") + '.</div><button class="btn" data-act="dismiss">Reopen the sheet</button></div>';
    }
    return "";
  }

  /* ---------------- render ---------------- */
  const stage = document.getElementById("stage");
  const wrap = document.getElementById("wrap");
  function backdrop() {
    if (state.form !== "B2") return "";
    const img = state.ctx === "town" ? "guild" : "battlemap";
    return '<div class="bd' + (state.ctx === "game" ? " map" : "") + '" style="background-image:url(art/' + img + '.jpg)"></div><div class="bd-shade"></div>';
  }
  function render(keepScroll) {
    const h = byKey[state.hero] || HEROES[0];
    const sc = {};
    if (keepScroll) ["lb-ab", "lb-ft"].forEach(id => { const el = document.getElementById(id); if (el) sc[id] = el.scrollTop; });
    const focusSearch = document.activeElement && document.activeElement.id === "fsearch";
    const caret = focusSearch ? document.activeElement.selectionStart : null;
    stage.className = "stage form-" + state.form + " ctx-" + state.ctx;
    stage.innerHTML = backdrop() + '<div class="sheet">' + identityCol(h) + mainCol(h) + "</div>" + overlay(h);
    if (keepScroll) Object.keys(sc).forEach(id => { const el = document.getElementById(id); if (el) el.scrollTop = sc[id]; });
    if (focusSearch) { const s = document.getElementById("fsearch"); if (s) { s.focus(); try { s.setSelectionRange(caret, caret); } catch (e) {} } }
    if (!keepScroll) stage.querySelectorAll(".sheet").forEach(s => { s.style.animation = "none"; void s.offsetWidth; s.style.animation = ""; });
    syncControls();
  }
  function syncControls() {
    document.querySelectorAll("#ctl-form button").forEach(b => b.classList.toggle("on", b.dataset.form === state.form));
    document.querySelectorAll("#ctl-party button").forEach(b => b.classList.toggle("on", b.dataset.party === state.party));
    document.querySelectorAll("#ctl-ctx button").forEach(b => b.classList.toggle("on", b.dataset.ctx === state.ctx));
    document.querySelectorAll("#ctl-zoom button").forEach(b => b.classList.toggle("on", b.dataset.zoom === state.zoom));
    const sel = document.getElementById("ctl-hero");
    if (sel.value !== state.hero) sel.value = state.hero;
  }
  function fit() {
    if (state.zoom === "full") {
      wrap.classList.add("full"); stage.style.transform = "none"; wrap.style.height = "auto";
    } else {
      wrap.classList.remove("full");
      const s = Math.min(1, wrap.clientWidth / 1920);
      stage.style.transform = "scale(" + s + ")"; wrap.style.height = Math.round(1080 * s) + "px";
    }
  }

  /* hero select */
  const sel = document.getElementById("ctl-hero");
  sel.innerHTML = HEROES.map(h => '<option value="' + esc(h.key) + '">' + esc(h.name) + " (level " + esc(h.level) + ")</option>").join("");
  sel.addEventListener("change", () => { setHero(sel.value); });
  function setHero(k) {
    state.hero = k; state.open = new Set([k + ":0"]); state.expanded = new Set(); state.overlay = null;
    if (!roster().includes(k) && state.party === "switch") { /* keep the hero even if outside the current list */ }
    render(false);
  }

  document.addEventListener("click", e => {
    const t = e.target.closest("[data-form],[data-party],[data-ctx],[data-zoom],[data-sim],[data-hero],[data-ab],[data-grp],[data-chip],[data-more],[data-act]");
    if (!t) return;
    if (t.dataset.form) { state.form = t.dataset.form; render(false); return; }
    if (t.dataset.party) { state.party = t.dataset.party; render(true); return; }
    if (t.dataset.ctx) { state.ctx = t.dataset.ctx; if (!roster().includes(state.hero)) state.hero = roster()[0]; render(false); return; }
    if (t.dataset.zoom) { state.zoom = t.dataset.zoom; syncControls(); fit(); return; }
    if (t.dataset.sim) { state.overlay = t.dataset.sim === "esc" ? "closed" : t.dataset.sim; render(true); return; }
    if (t.dataset.hero) { setHero(t.dataset.hero); return; }
    if (t.dataset.ab && e.target.closest(".art")) { const id = t.dataset.ab; state.open.has(id) ? state.open.delete(id) : state.open.add(id); render(true); return; }
    if (t.dataset.grp) { const g = t.dataset.grp; state.closedGroups.has(g) ? state.closedGroups.delete(g) : state.closedGroups.add(g); render(true); return; }
    if (t.dataset.chip) { const c = t.dataset.chip; state.chips.has(c) ? state.chips.delete(c) : state.chips.add(c); render(true); return; }
    if (t.dataset.more) { const id = t.dataset.more; state.expanded.has(id) ? state.expanded.delete(id) : state.expanded.add(id); render(true); return; }
    const act = t.dataset.act;
    if (act === "showall") { state.showAll = !state.showAll; render(true); }
    else if (act === "clear") { state.chips.clear(); state.search = ""; render(true); }
    else if (act === "dismiss") { state.overlay = null; render(true); }
    else if (act === "close") { state.overlay = "closed"; render(true); }
    else if (act === "appearance" || act === "builder") { flash(t, act === "appearance" ? "Opens the builder's Appearance page" : "Opens the builder (before the first win)"); }
  });
  document.addEventListener("input", e => { if (e.target.id === "fsearch") { state.search = e.target.value; render(true); } });
  document.addEventListener("keydown", e => {
    if (e.key === "Escape") { state.overlay = state.overlay === "closed" ? null : "closed"; render(true); }
  });
  function flash(btn, text) {
    const old = btn.innerHTML; btn.textContent = text; btn.disabled = true;
    setTimeout(() => { btn.innerHTML = old; btn.disabled = false; }, 1400);
  }
  window.addEventListener("resize", fit);

  //a bare #A / #B1 / #B2 in the link opens that form directly.
  const hp = (location.hash || "").replace("#", "").split(",");
  if (hp[0]) state.form = hp[0]; if (hp[1]) state.ctx = hp[1]; if (hp[2]) state.hero = hp[2];
  state.open = new Set([state.hero + ":0"]);
  render(false); fit();
})();
</script>
